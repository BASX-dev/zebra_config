// Force background color to red
(function () {
    function setRedBackground() {
        document.documentElement.style.backgroundColor = "red";
        document.body.style.backgroundColor = "red";
    }

    // Run immediately
    setRedBackground();

    // Run again after DOM loads (in case the page overrides it)
    document.addEventListener("DOMContentLoaded", setRedBackground);

    // Observe for dynamic changes and reapply
    const observer = new MutationObserver(() => {
        setRedBackground();
    });

    observer.observe(document.documentElement, {
        childList: true,
        subtree: true
    });
})();
